# TypeScript Kernel Implementation Plan

**Goal:** A browser + Node TypeScript package that reproduces all 51 frozen vectors (`test-vectors/*.json`).

**Architecture:** A byte-exact kernel with no runtime dependencies beyond `@noble/ed25519` / `@noble/hashes`. Modules: strict JSON parser + canonical serializer, Ed25519 key/signature encoding, JSON Schema subset, the pure §10.3 decision, and the §26.3 state machine. A vector runner proves conformance before any role/API work.

**Tech Stack:** TypeScript 5, ESM, Vitest, `@noble/ed25519` v2, `@noble/hashes`.

**Spec:** `CCP_Working_Draft_v0.5.md`; contract: `test-vectors/`.

## Global Constraints

- Reproduce every vector in `test-vectors/*.json` exactly.
- No floats, negatives, `NaN`, `Infinity`; duplicate keys rejected; NFC strings/keys; keys sorted by Unicode code point; `signature` last.
- Ed25519: 32-byte seed; `key:ed25519:<base64url>`; `ed25519:<base64url>`; base64url unpadded.
- `src/` must run in both browser and Node — no `fs`, `path`, `Buffer`, or `process` in `src/`.
- Commit only when the user asks.

`sdk/typescript/src/json.ts`:
```ts
export type JsonValue =
  | null
  | boolean
  | string
  | bigint
  | JsonValue[]
  | { [key: string]: JsonValue };

const SHORT: Record<string, string> = {
  "\b": "\\b", "\f": "\\f", "\n": "\\n", "\r": "\\r", "\t": "\\t",
};

export function quote(value: string): string {
  const s = value.normalize("NFC");
  let out = '"';
  for (const ch of s) {
    const code = ch.codePointAt(0)!;
    if (ch === '"') out += '\\"';
    else if (ch === "\\") out += "\\\\";
    else if (SHORT[ch]) out += SHORT[ch];
    else if (code < 0x20) out += "\\u" + code.toString(16).padStart(4, "0");
    else out += ch;
  }
  return out + '"';
}

function compareCodePoint(a: string, b: string): number {
  const aa = Array.from(a);
  const bb = Array.from(b);
  const n = Math.min(aa.length, bb.length);
  for (let i = 0; i < n; i++) {
    const diff = aa[i].codePointAt(0)! - bb[i].codePointAt(0)!;
    if (diff !== 0) return diff;
  }
  return aa.length - bb.length;
}

export function serialize(value: JsonValue): string {
  if (value === null) return "null";
  if (value === true) return "true";
  if (value === false) return "false";
  if (typeof value === "string") return quote(value);
  if (typeof value === "bigint") {
    if (value < 0n) throw new Error("negative");
    return value.toString();
  }
  if (Array.isArray(value)) {
    return "[" + value.map(serialize).join(",") + "]";
  }
  if (typeof value === "object") {
    const keys = Object.keys(value)
      .filter((key) => key !== "signature")
      .sort(compareCodePoint);
    const parts = keys.map((key) => quote(key) + ":" + serialize(value[key]));
    if ("signature" in value) {
      parts.push(quote("signature") + ":" + serialize(value.signature));
    }
    return "{" + parts.join(",") + "}";
  }
  throw new TypeError("unsupported type");
}
```

```ts
import type { JsonValue } from "./json.js";

export function parseStrict(text: string): JsonValue {
  let i = 0;
  const fail = (code: string): never => { throw new SyntaxError(code); };
  const skipWs = () => {
    while (i < text.length && " \t\n\r".includes(text[i])) i++;
  };
  const expect = (token: string) => {
    if (text.slice(i, i + token.length) !== token) fail("malformed");
    i += token.length;
  };
  const parseString = (): string => {
    i++; // opening quote
    let out = "";
    while (i < text.length) {
      const c = text[i++];
      if (c === '"') return out;
      if (c === "\\") {
        const e = text[i++];
        if (e === '"') out += '"';
        else if (e === "\\") out += "\\";
        else if (e === "/") out += "/";
        else if (e === "b") out += "\b";
        else if (e === "f") out += "\f";
        else if (e === "n") out += "\n";
        else if (e === "r") out += "\r";
        else if (e === "t") out += "\t";
        else if (e === "u") {
          const hex = text.slice(i, i + 4);
          if (!/^[0-9a-fA-F]{4}$/.test(hex)) fail("malformed");
          const cp = parseInt(hex, 16);
          i += 4;
          if (cp >= 0xd800 && cp <= 0xdbff && text[i] === "\\" && text[i + 1] === "u") {
            const lo = parseInt(text.slice(i + 2, i + 6), 16);
            i += 6;
            out += String.fromCodePoint(0x10000 + ((cp - 0xd800) << 10) + (lo - 0xdc00));
          } else {
            out += String.fromCodePoint(cp);
          }
        } else fail("malformed");
      } else if (c.charCodeAt(0) < 0x20) {
        fail("malformed");
      } else {
        out += c;
      }
    }
    return fail("malformed");
  };
  const parseNumber = (): bigint => {
    if (text[i] === "-") fail("negative");
    const start = i;
    while (i < text.length && text[i] >= "0" && text[i] <= "9") i++;
    if (i === start) fail("malformed");
    if (text[i] === "." || text[i] === "e" || text[i] === "E") fail("float");
    const digits = text.slice(start, i);
    if (digits.length > 1 && digits[0] === "0") fail("malformed");
    return BigInt(digits);
  };
  const parseValue = (): JsonValue => {
    skipWs();
    const c = text[i];
    if (c === "{") return parseObject();
    if (c === "[") return parseArray();
    if (c === '"') return parseString();
    if (c === "-") return fail("negative");
    if (c === "t") { expect("true"); return true; }
    if (c === "f") { expect("false"); return false; }
    if (c === "n") { expect("null"); return null; }
    if (c >= "0" && c <= "9") return parseNumber();
    return fail("malformed");
  };
  const parseObject = (): { [key: string]: JsonValue } => {
    i++; // {
    const obj: { [key: string]: JsonValue } = {};
    const seen = new Set<string>();
    skipWs();
    if (text[i] === "}") { i++; return obj; }
    for (;;) {
      skipWs();
      if (text[i] !== '"') fail("malformed");
      const key = parseString();
      if (seen.has(key)) fail("duplicate");
      seen.add(key);
      skipWs();
      expect(":");
      obj[key] = parseValue();
      skipWs();
      if (text[i] === ",") { i++; continue; }
      if (text[i] === "}") { i++; return obj; }
      fail("malformed");
    }
  };
  const parseArray = (): JsonValue[] => {
    i++; // [
    const arr: JsonValue[] = [];
    skipWs();
    if (text[i] === "]") { i++; return arr; }
    for (;;) {
      arr.push(parseValue());
      skipWs();
      if (text[i] === ",") { i++; continue; }
      if (text[i] === "]") { i++; return arr; }
      fail("malformed");
    }
  };
  const result = parseValue();
  skipWs();
  if (i !== text.length) fail("malformed");
  return result;
}
```

```ts
import * as ed from "@noble/ed25519";
import { sha512 } from "@noble/hashes/sha512";

ed.etc.sha512Sync = (...messages: Uint8Array[]) =>
  sha512(ed.etc.concatBytes(...messages));

const KEY_PREFIX = "key:ed25519:";
const SIG_PREFIX = "ed25519:";

export function b64urlEncode(bytes: Uint8Array): string {
  let binary = "";
  for (const byte of bytes) binary += String.fromCharCode(byte);
  return btoa(binary).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

export function b64urlDecode(text: string): Uint8Array {
  const base = text.replace(/-/g, "+").replace(/_/g, "/");
  const padded = base + "=".repeat((4 - (base.length % 4)) % 4);
  const binary = atob(padded);
  return Uint8Array.from(binary, (char) => char.charCodeAt(0));
}

export function keyIdFromPublic(raw: Uint8Array): string {
  return KEY_PREFIX + b64urlEncode(raw);
}

export function publicBytesFromKeyId(keyId: string): Uint8Array {
  if (!keyId.startsWith(KEY_PREFIX)) throw new Error("unsupported key id");
  return b64urlDecode(keyId.slice(KEY_PREFIX.length));
}

export function publicFromSeed(seed: Uint8Array): Uint8Array {
  if (seed.length !== 32) throw new Error("seed must be 32 bytes");
  return ed.getPublicKey(seed);
}

export function sign(message: Uint8Array, seed: Uint8Array): string {
  return SIG_PREFIX + b64urlEncode(ed.sign(message, seed));
}

export function verify(message: Uint8Array, signatureField: string, publicKey: Uint8Array): boolean {
  if (!signatureField.startsWith(SIG_PREFIX)) return false;
  return ed.verify(b64urlDecode(signatureField.slice(SIG_PREFIX.length)), message, publicKey);
}
```
